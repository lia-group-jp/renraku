<div id="login">
  <input id="pass" type="password" autocomplete="off" placeholder="パスワード">
  <button id="loginBtn" onclick="login()">入る</button>
  <div class="err" id="loginErr"></div>
</div>

<div id="chat">
  <div id="list"></div>
  <form onsubmit="send(event)">
    <input id="name" list="roster" placeholder="あなたの名前を選ぶ(文字を打つと絞り込み)" autocomplete="off" onchange="nameChanged()">
    <datalist id="roster"></datalist>
    <div class="line">
      <textarea id="body" placeholder="メッセージを入力" maxlength="2000"></textarea>
      <button id="sendBtn">送信</button>
    </div>

<script>
  const $ = id => document.getElementById(id);
  const key = s => String(s).replace(/[\s　]/g, '');   // 全角・半角の空白を無視して名前を比べる
  // 保存できない環境(プライベートブラウズ等)でも画面は動かす
  const keep = (k, v) => {
    try {
      if (v != null) sessionStorage.setItem(k, v);
      return sessionStorage.getItem(k) || '';
    } catch (e) { return ''; }
  };
  let pass = keep('pass');
  let all = [];     // 全発言

  function call(action, name, body) {
    return new Promise((ok, ng) => google.script.run.withSuccessHandler(ok).withFailureHandler(ng).api(pass, action, name, body));
  }

  function fail(e) {
    // GAS側の英語エラー(同時実行超過など)は、そのまま出しても社員に伝わらない
    const msg = /[^\x00-\x7F]/.test(String(e.message)) ? e.message : '通信に失敗しました。少し待ってからやり直してください';
    if (String(e.message).includes('PASS')) {
      keep('pass', '');
      $('chat').style.display = 'none';
      $('login').style.display = 'block';
      $('pass').value = ''; // 古いパスワードが残ったままだと打ち直せないので消す
      $('loginErr').textContent = 'パスワードが違います';
    } else if ($('chat').style.display !== 'flex') {
      // ログイン前のエラーも見えるようにする(見えないと「押しても何も起きない」になる)
      $('loginErr').textContent = msg;
    } else {
      $('err').textContent = msg;
    }
  }

  const el = (tag, cls, text) => {
    const e = document.createElement(tag);
    if (cls) e.className = cls;
    if (text != null) e.textContent = text;
    return e;
  };

  // full=全件、それ以外は最新分。最新分を全体の後ろにはめ直す(シートで直した内容もそのまま出る)
  function apply(data) {
    if (data.full) {
      all = data.msgs;
    } else {
      const k = data.total - data.msgs.length;
      if (k > all.length) return call('all').then(apply);   // 取りこぼした時は全部読み直す
      all = all.slice(0, Math.max(0, k)).concat(data.msgs);
    }
    render(data.roster);
  }

  function render(roster) {
    $('roster').replaceChildren(...roster.map(n => Object.assign(el('option'), { value: n })));
    const list = $('list');
    const atBottom = list.scrollHeight - list.scrollTop - list.clientHeight < 60;
    const me = key($('name').value.trim());
    const items = [];
    let lastDay = '';
    all.forEach(([time, who, text]) => {
      if (text === '') return;
      // 日時は「yyyy/MM/dd HH:mm」。日付が変わったら区切りを入れ、本文には時刻だけ出す
      const [day, hm = ''] = String(time).split(' ');
      if (day !== lastDay) {
        const d = el('div', 'day');
        d.append(el('span', null, day));
        items.push(d);
        lastDay = day;
      }
      const mine = me !== '' && key(who) === me;
      const row = el('div', 'row' + (mine ? ' me' : ''));
      if (mine) {
        // 自分の発言。吹き出しの左下に「既読」と時刻(LINEと同じ並び)
        const meta = el('div', 'meta');
        meta.append(el('div', 'read', '既読'), el('div', 'time', hm));
        row.append(meta, el('div', 'bubble', text));
      } else {
        // 相手の発言。アイコンは名前の1文字目、時刻は吹き出しの右下
        const col = el('div', 'col');
        const line = el('div', 'line');
        line.append(el('div', 'bubble', text), el('div', 'time', hm));
        col.append(el('div', 'who', who), line);
        row.append(el('div', 'avatar', String(who).trim().charAt(0)), col);
      }
      items.push(row);
    });
    list.replaceChildren(...items);
    if (atBottom || !render.done) list.scrollTop = list.scrollHeight;
    render.done = true;
  }

  // 名前を選び直したら、自分の発言と既読の出方が変わるので描き直す
  function nameChanged() {
    keep('name', $('name').value.trim());
    refresh();
    render.done = false;
    if (all.length) render([...$('roster').options].map(o => o.value));
  }

  // 前の読み込みが終わる前に次を投げない(遅れて返った古い内容で上書きされるのを防ぐ)
  let refreshing = false;
  function refresh() {
    if (refreshing || document.hidden || $('chat').style.display !== 'flex') return;
    refreshing = true;
    return call('list').then(apply).catch(fail).finally(() => { refreshing = false; });
  }

  function login() {
    pass = $('pass').value.trim();
    $('loginBtn').disabled = true;
    $('loginErr').textContent = '読み込んでいます…';
    // 最初に全件読む。件数が多いと数秒かかる
    call('all').then(data => {
      keep('pass', pass);
      $('login').style.display = 'none';
      $('chat').style.display = 'flex';
      $('loginErr').textContent = '';
      apply(data);
    }).catch(fail).finally(() => { $('loginBtn').disabled = false; });
  }

  function send(ev) {
    ev.preventDefault();
    $('err').textContent = '';
    $('sendBtn').disabled = true;
    call('post', $('name').value, $('body').value)
      .then(data => { $('body').value = ''; render.done = false; apply(data); })
      .catch(fail)
      .finally(() => { $('sendBtn').disabled = false; });
  }

  setInterval(refresh, 30000);
  document.addEventListener('visibilitychange', refresh);
  $('name').value = keep('name');   // 前に選んだ名前を出しておく(既読を付けるため)
  if (pass) { $('pass').value = pass; login(); }
</script>
